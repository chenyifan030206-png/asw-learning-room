# 构建工作清单：主课默认只有timing库、timeline_demo和timeline_tests。
cmake_minimum_required(VERSION 3.16)
project(asw_timeline LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Werror)
add_library(timing STATIC timing.c)
target_link_libraries(timing PUBLIC m)
target_include_directories(timing PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(timeline_demo demo.c)
target_link_libraries(timeline_demo PRIVATE timing)
include(CTest)
if(BUILD_TESTING)
    add_executable(timeline_tests test_timing.c)
    target_link_libraries(timeline_tests PRIVATE timing)
    add_test(NAME timeline_contract COMMAND timeline_tests)
endif()
# 新内容放附录：默认OFF，不影响主课。需要补原D08时再显式ON。
option(BUILD_APPENDIX "Build original D08 signal appendix" OFF)
if(BUILD_APPENDIX)
    add_library(signal_check STATIC signal.c)
    target_include_directories(signal_check PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(signal_check PUBLIC m)
    add_executable(signal_demo demo_signal.c)
    target_link_libraries(signal_demo PRIVATE signal_check)
    if(BUILD_TESTING)
        add_executable(signal_tests test_signal.c)
        target_link_libraries(signal_tests PRIVATE signal_check)
        add_test(NAME signal_contract COMMAND signal_tests)
    endif()
endif()
